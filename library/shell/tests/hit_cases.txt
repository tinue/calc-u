# The cases of the hit rule (LIB-14 spec AC-24, EC-8; design §3.7, H1 to H3,
# D-8), read by library/shell/tests/hit_map_tests.cpp and
# library/web/tests/shell_model.test.js alike.
#
#   map <distance> <key>:<x>,<y>,<w>,<h> ...   a hit map, keys in its order
#   at <scale> <px> <py> <key or ->            the key found, or none (-)
#
# Every `at` line asks the last `map`.

# Three keys of 10 x 10 units in a row, 10 apart; 4 logical pixels.
map 4 a:0,0,10,10 b:20,0,10,10 c:40,0,10,10
at 1 5 5 a
at 1 0 0 a
at 1 10 10 a
at 1 25 5 b
at 1 13 5 a
at 1 14 5 a
at 1 14.5 5 -
at 1 16 5 b
at 1 15 5 -
at 1 36 5 c
at 1 5 13 a
at 1 5 14.1 -
at 1 -4 -1 -
at 1 -3 0 a
at 1 60 60 -
# Scaled 2x: the areas grow, the distance does not.
at 2 21 10 a
at 2 24 10 a
at 2 25 10 -
at 2 36 10 b
at 2 50 10 b

# Overlapping areas: the first in the hit map wins.
map 0 x:0,0,20,20 y:10,10,20,20
at 1 15 15 x
at 1 25 25 y
at 1 31 31 -

# Distance 0 (Calc-U-59's widened areas): only inside.
map 0 k:0,0,10,10 l:12,0,10,10
at 1 10 5 k
at 1 11 5 -
at 1 12 5 l

# Equal distance: the first.
map 10 p:0,0,10,10 q:20,0,10,10
at 1 15 5 p
at 1 15.5 5 q
